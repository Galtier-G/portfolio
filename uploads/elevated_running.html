def compute_distance(p1, p2):
    """ Compute the distance between two GPS points using geodesic distance.

    Args:
        p1 int : point 1  
        p2 int: point 2

    Returns:
        float: distance in meters
    """ 
    return geodesic((p1['latitude'], p1['longitude']), (p2['latitude'], p2['longitude'])).meters

def compute_pace(speed):
    """ Compute the pace in minutes per kilometer from speed in meters per second.
    
    Args:
        speed float: speed in meters per second
    Returns:
        float : pace in minutes per kilometer
    """
    return 1000 / speed / 60 if speed > 0 else np.nan  # min/km

# will store the segments
segments = []

for activity_id, group in df.groupby('activity_id'): 
    group = group.reset_index(drop=True)
    
    i = 0
    while i < len(group) - 1:
        dist = 0
        j = i + 1
        while j < len(group) and dist < SEGMENT_SIZE:
            dist += compute_distance(group.loc[j-1], group.loc[j]) 
            j += 1
        
        if dist >= SEGMENT_SIZE: 
            alt_diff = group.loc[j-1, 'altitude'] - group.loc[i, 'altitude'] # altitude difference in meters for the segment
            slope = (alt_diff / dist) * 100  # slope claculation in percentage
            segment_duration = (pd.to_datetime(group.loc[j-1, 'timestamp']) - pd.to_datetime(group.loc[i, 'timestamp'])).total_seconds()
            speed = dist / segment_duration if segment_duration > 0 else np.nan
            pace = compute_pace(speed)
            
            if -20 <= slope <= 20: # filter segments with slope between -20% and 20% to normalize the data 
                segments.append({
                    'activity_id': activity_id,
                    'slope': slope,
                    'pace': pace
                })
        i = j # move to the next segment start point

def format_seconds(s):
    """Format seconds into HH:MM:SS format.
    Args:
        s (float): Number of seconds.
    Returns:    
        str: Formatted time string in HH:MM:SS format."""
    total_seconds = round(s)
    hours, remainder = divmod(total_seconds, 3600)
    minutes, seconds = divmod(remainder, 60)
    return f"{hours:02}:{minutes:02}:{seconds:02}"


GPX_PATH = "C:\\Users\\germain.galtier\\code\\dataset_test_pro\\data\\raw\\gpx\\trail-du-saleve-2025-la-thuile1742297031233.gpx" 

SEGMENT_SIZE = 1000  # bin size for grouping GPS points in meters

# load GPX file
with open(GPX_PATH, 'r') as gpx_file:
    gpx = gpxpy.parse(gpx_file)

# Points extraction from GPX
points = []
for track in gpx.tracks:
    for segment in track.segments:
        for p in segment.points:
            points.append({
                'latitude': p.latitude,
                'longitude': p.longitude,
                'elevation': p.elevation
            })

df_gpx = pd.DataFrame(points) # initialize a DataFrame with the GPX points

# segmentation
segments = []
i = 0
while i < len(df_gpx) - 1:
    dist = 0
    j = i + 1
    while j < len(df_gpx) and dist < SEGMENT_SIZE:
        p1 = df_gpx.loc[j - 1]
        p2 = df_gpx.loc[j]
        dist += geodesic((p1['latitude'], p1['longitude']), (p2['latitude'], p2['longitude'])).meters
        j += 1

    if dist >= SEGMENT_SIZE:
        start = df_gpx.loc[i]
        end = df_gpx.loc[j - 1]
        elevation_gain = end['elevation'] - start['elevation']
        slope = (elevation_gain / dist) * 100
        slope_bin = int(round(slope))
        slope_bin = min(20, max(-20, slope_bin))  # clip between -20 and +20

        # usage of the quadratic regression model to predict the pace
        row = df_quadratic[df_quadratic['slope_%'] == slope_bin]
        if not row.empty:
            pace_min_per_km = row['predicted_pace_min_per_km'].values[0]
            pace_sec_per_m = (pace_min_per_km * 60) / 1000
            segment_time_sec = pace_sec_per_m * dist

            segments.append({
                'start_index': i,
                'end_index': j - 1,
                'distance_m': dist,
                'slope_%': slope,
                'slope_bin': slope_bin,
                'pace_min_per_km': pace_min_per_km,
                'segment_time_s': segment_time_sec
            })
    i = j

df_segments = pd.DataFrame(segments) # writing the data into a DataFrame

# cumulative sum calculations
df_segments['cum_distance_m'] = df_segments['distance_m'].cumsum()
df_segments['cum_time_s'] = df_segments['segment_time_s'].cumsum()
df_segments['km'] = (df_segments['cum_distance_m'] / 1000).astype(int)

# km grouping
df_km = df_segments.groupby('km').agg({
    'segment_time_s': 'sum',
    'slope_%': 'mean'  # sloap average for each km
}).reset_index()

df_km.rename(columns={'slope_%': 'slope_mean'}, inplace=True)

df_km['segment_time'] = df_km['segment_time_s'].apply(format_seconds)

df_km['slope_mean'] = df_km['slope_mean'].round(2)
